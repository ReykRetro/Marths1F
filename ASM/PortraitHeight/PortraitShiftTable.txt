// ============================================================
// Portrait Shift Table (version tabla de bytes, en espacio libre)
// Basado en el parche de Portrait Height ("Download this one").
//
// 1 byte por ID de retrato:  0 = normal,  distinto de 0 = retrato desplazado hacia arriba.
// Esta tabla cubre los IDs 0x00 - 0xFF (256 retratos).
// Para marcar un retrato usa ShiftPortrait(ID) mas abajo.
// ============================================================

// Rutina que reemplaza la original en 0x5C24 (lee el byte de la tabla).
//   ldr  r1, [pc, #4]   ; r1 = puntero a la tabla (el POIN de abajo)
//   ldrb r0, [r1, r0]   ; r0 = tabla[ID del retrato]
//   bx   lr
//   nop
PUSH
ORG 0x5C24
BYTE $01 $49 $08 $5C $70 $47 $C0 $46
POIN PortraitShiftEdit
POP

// La tabla va en espacio libre (por eso no hace falta que quepa en la rutina original).
ALIGN 4
PortraitShiftEdit:
BYTE 0 0 0 1 1 0 0 0  // 0x00-0x07
BYTE 0 0 0 0 0 0 0 0  // 0x08-0x0F
BYTE 0 0 0 0 0 0 0 0  // 0x10-0x17
BYTE 1 0 0 0 0 0 0 0  // 0x18-0x1F
BYTE 0 0 0 0 0 0 0 1  // 0x20-0x27
BYTE 0 0 0 0 0 0 1 0  // 0x28-0x2F
BYTE 0 0 0 0 0 0 0 1  // 0x30-0x37
BYTE 0 0 0 0 1 0 0 0  // 0x38-0x3F
BYTE 0 0 0 0 0 0 0 0  // 0x40-0x47
BYTE 0 0 0 0 0 0 0 0  // 0x48-0x4F
BYTE 0 0 0 0 0 0 1 0  // 0x50-0x57
BYTE 0 0 0 0 0 0 0 0  // 0x58-0x5F
BYTE 0 0 0 0 0 0 0 0  // 0x60-0x67
BYTE 0 0 0 0 0 0 0 0  // 0x68-0x6F
BYTE 0 0 0 0 0 0 0 0  // 0x70-0x77
BYTE 0 0 0 0 0 0 0 0  // 0x78-0x7F
BYTE 0 0 0 0 0 0 0 0  // 0x80-0x87
BYTE 0 0 0 0 0 0 0 0  // 0x88-0x8F
BYTE 0 0 0 0 0 0 0 0  // 0x90-0x97
BYTE 0 0 0 0 0 0 0 0  // 0x98-0x9F
BYTE 0 0 0 0 0 0 0 0  // 0xA0-0xA7
BYTE 0 0 0 0 0 0 0 0  // 0xA8-0xAF
BYTE 0 0 0 0 0 0 0 0  // 0xB0-0xB7
BYTE 0 0 0 0 0 0 0 0  // 0xB8-0xBF
BYTE 0 0 0 0 0 0 0 0  // 0xC0-0xC7
BYTE 0 0 0 0 0 0 0 0  // 0xC8-0xCF
BYTE 0 0 0 0 0 0 0 0  // 0xD0-0xD7
BYTE 0 0 0 0 0 0 0 0  // 0xD8-0xDF
BYTE 0 0 0 0 0 0 0 0  // 0xE0-0xE7
BYTE 0 0 0 0 0 0 0 0  // 0xE8-0xEF
BYTE 0 0 0 0 0 0 0 0  // 0xF0-0xF7
BYTE 0 0 0 0 0 0 0 0  // 0xF8-0xFF

// Pone el byte de un retrato en 1 (desplazado hacia arriba).
#define ShiftPortrait(id) "PUSH; ORG PortraitShiftEdit+id; BYTE 1; POP"

// ---- Retratos desplazados hacia arriba ----
ShiftPortrait(0x02) // Marth
ShiftPortrait(0x03) // Marth (segundo retrato, bit 0x800000)
